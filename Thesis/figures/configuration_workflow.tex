\begin{tikzpicture}[
  font=\scriptsize,
  input/.style={draw, dashed, fill=white, rounded corners=2pt,
                align=center, minimum width=30mm, minimum height=10mm},
  process/.style={draw, fill=white, rounded corners=2pt,
                  align=center, minimum width=32mm, minimum height=12mm},
  output/.style={draw, fill=black!8, rounded corners=2pt,
                 align=center, minimum width=30mm, minimum height=10mm},
  flow/.style={-{Latex[length=2mm]}, line width=.7pt}
]
\node[input] (features) {Feature configuration\\(design choices)};
\node[input, below=5mm of features] (deployment) {Deployment specification\\(hosts, roles, links)};
\node[input, below=5mm of deployment] (plugins) {Property plug-ins\\(class paths and settings)};
\node[process, right=14mm of deployment] (generator) {Configuration\\generator};
\node[output, right=14mm of generator, yshift=10mm] (yaml1) {Host A runtime YAML};
\node[output, right=14mm of generator, yshift=-10mm] (yaml2) {Host B runtime YAML};
\node[process, right=13mm of yaml1] (monitor) {Monitor runtime};
\node[process, right=13mm of yaml2] (runner) {Verifier runtime};

\foreach \n in {features,deployment,plugins}
  \draw[flow] (\n.east) -- (generator.west);
\draw[flow] (generator) -- (yaml1);
\draw[flow] (generator) -- (yaml2);
\draw[flow] (yaml1) -- (monitor);
\draw[flow] (yaml2) -- (runner);
\end{tikzpicture}
