\section{Config Generator}

The generator creates ROS2-Monitor-Infra runtime YAML from a JSON generation
request. The request is produced after resource, class, transport, and
placement choices have been made. It explicitly describes the runtime
topology, so the generator derives the YAML files from runtimes and links.

\begin{enumerate}
  \item discover visible ROS2 topics, services, and actions;
  \item select the resources to monitor;
  \item select custom converter and verdict classes;
  \item describe runtime locations and links in a JSON request;
  \item project the JSON request into one or more runtime YAML files.
\end{enumerate}

\section{Generation Request JSON}

The JSON request is an intermediate representation between user choices and
runtime YAML. It separates reusable monitoring elements from deployment
topology.

\begin{tabular}{@{}p{0.20\linewidth}p{0.73\linewidth}@{}}
\toprule
Section & Meaning \\
\midrule
\texttt{observations} &
ROS2 resources to observe. Each item is a topic, service, or action with a
stable id, ROS name, interface type, and optional framework settings. \\
\texttt{monitors} &
Monitoring chains. Each item contains one converter class, one verdict class,
their parameters, and the observation or monitor ids it consumes. \\
\texttt{runtimes} &
Generated runtime files. Each runtime declares which observations it captures,
which monitors it hosts, and where its output is written. \\
\texttt{links} &
Typed edges between runtimes. A link declares source runtime, target runtime,
payload type, transport, and carried observations or monitors. \\
\bottomrule
\end{tabular}

\subsection{Abstract schema}

\begin{verbatim}
{
  "observations": [
    {
      "id": "<observation id>",
      "kind": "topic | service | action",
      "name": "<ROS2 name>",
      "type": "<ROS2 interface type>",
      "transformers": [],
      "qos": "<optional>",
      "phases": "<optional>"
    }
  ],

  "monitors": [
    {
      "id": "<monitor id>",
      "converter": {
        "type": "<module.path:ClassName>",
        "params": {}
      },
      "verdict": {
        "type": "<module.path:ClassName>",
        "params": {}
      },
      "consumes": ["<observation id | monitor id>"]
    }
  ],

  "runtimes": [
    {
      "id": "<runtime id>",
      "file": "<output yaml file>",
      "output": {
        "output_dir": "<path>",
        "session_id_prefix": "<prefix>"
      },
      "captures": ["<observation id>"],
      "hosts": ["<monitor id>"]
    }
  ],

  "links": [
    {
      "id": "<link id>",
      "from": "<runtime id>",
      "to": "<runtime id>",
      "payload": "records | verdicts",
      "transport": {
        "type": "mqtt | file",
        "<setting>": "<value>"
      },
      "carries": ["<observation id | monitor id>"]
    }
  ]
}
\end{verbatim}

\section{Config Generation Algorithm}

\begin{algorithm}[H]
\caption{Generate runtime YAML from JSON}
\begin{algorithmic}[1]
\Require generation request $J$
\Ensure generated YAML files $F$
\State $F \gets \emptyset$
\ForAll{runtime $R$ in $J.\text{runtimes}$}
  \State $Y \gets \Call{NewYAML}{}$
  \State $Y.\text{monitor} \gets R.\text{output}$
  \ForAll{observation id $o$ in $R.\text{captures}$}
    \State $obs \gets J.\text{observations}[o]$
    \State $entry \gets \Call{MakeSourceEntry}{obs}$
    \State add $entry$ to $Y[obs.\text{kind}]$
  \EndFor
  \ForAll{incoming link $e$ to $R$}
    \State $Y.\text{verdict\_runner.source} \gets \Call{MakeSource}{e.\text{transport},e.\text{payload}}$
  \EndFor
  \ForAll{monitor id $m$ in $R.\text{hosts}$}
    \State $mon \gets J.\text{monitors}[m]$
    \State $chain \gets \Call{MakeConverterChain}{mon}$
    \State $chain.\text{inputs} \gets \Call{ResolveInputs}{mon.\text{consumes},J,R}$
    \State $chain.\text{verdict.exporters} \gets \Call{ResolveVerdictExporters}{m,R,J}$
    \State add $chain$ to $Y.\text{converters}$
  \EndFor
  \ForAll{outgoing link $e$ from $R$ where $e.\text{payload}=\text{records}$}
    \State \Call{AttachRecordExporter}{Y,e}
  \EndFor
  \State $F[R.\text{file}] \gets Y$
\EndFor
\State \Call{WriteYAMLFiles}{F}
\State \Return $F$
\end{algorithmic}
\end{algorithm}

\section{Runtime Resource Scanning}

\begin{algorithm}[H]
\caption{Scan ROS2 resources}
\begin{algorithmic}[1]
\Ensure resource catalogue $G$
\State $T \gets$ run \texttt{ros2 topic list -t}
\State $V \gets$ run \texttt{ros2 service list -t}
\State $A \gets$ run \texttt{ros2 action list -t}
\State $G \gets \emptyset$
\ForAll{topic $(name,type)$ in $T$}
  \State add $(topic,name,type)$ to $G$
\EndFor
\ForAll{service $(name,type)$ in $V$}
  \State add $(service,name,type)$ to $G$
\EndFor
\ForAll{action $(name,type)$ in $A$}
  \State add $(action,name,type)$ to $G$
\EndFor
\State \Return sorted $G$
\end{algorithmic}
\end{algorithm}

\section{Projection Rules}

\subsection{Observations}

Each captured observation becomes a source entry in the runtime that captures
it:

\begin{itemize}
  \item \texttt{topic} observations become entries under \texttt{topics};
  \item \texttt{service} observations become entries under \texttt{services};
  \item \texttt{action} observations become entries under \texttt{actions}.
\end{itemize}

\subsection{Monitors}

Each hosted monitor becomes one \texttt{converters} entry. Converter class,
verdict class, and parameters are copied from the JSON request. The
\texttt{inputs} field is resolved from the monitor's \texttt{consumes} ids.
For observations, it becomes the selected ROS2 source names. For aggregation
monitors, it refers to incoming verdict records.

\subsection{Links}

Links produce inter-runtime wiring:

\begin{itemize}
  \item outgoing \texttt{records} links add DataRecord exporters to the
        producing runtime;
  \item incoming \texttt{records} links add \texttt{verdict\_runner.source}
        to the consuming runtime;
  \item outgoing \texttt{verdicts} links add verdict exporters to the carried
        hosted monitors;
  \item incoming \texttt{verdicts} links are consumed by aggregation monitors
        through a verdict-aware source.
\end{itemize}

\begin{verbatim}
# record link: producing runtime
exporters:
  - type: <link.transport.type>
    <link.transport settings>

# record link: consuming runtime
verdict_runner:
  source:
    type: <link.transport.type>
    <link.transport settings>
\end{verbatim}

\section{Concrete Examples}

\subsection{Traditional}

\begin{verbatim}
JSON request:
{
  "observations": [
    {"id": "odom", "kind": "topic", "name": "/odom",
     "type": "nav_msgs/msg/Odometry"}
  ],
  "monitors": [
    {"id": "speed", "converter": {"type": "custom.odom_speed_converter:OdomSpeedConverter",
     "params": {}}, "verdict": {"type": "custom.odom_speed_verdict:OdomSpeedVerdict",
     "params": {}}, "consumes": ["odom"]}
  ],
  "runtimes": [
    {"id": "local", "file": "config.yaml",
     "output": {"output_dir": "/output", "session_id_prefix": "local"},
     "captures": ["odom"], "hosts": ["speed"]}
  ],
  "links": []
}

Generated config.yaml:
monitor:
  output_dir: /output
  session_id_prefix: local

topics:
  - name: /odom
    type: nav_msgs/msg/Odometry

converters:
  - type: custom.odom_speed_converter:OdomSpeedConverter
    inputs: [/odom]
    verdict:
      type: custom.odom_speed_verdict:OdomSpeedVerdict
      exporters:
        - type: stdout
\end{verbatim}
